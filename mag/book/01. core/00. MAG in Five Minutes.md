# MAG in Five Minutes

```mag
// Map operations collide with general built-ins such as get and count, so this
// guide gives that owning module a short namespace alias.
import core.map as maps

// Two slashes start a comment. Calls put the function first, followed by
// comma-separated arguments: function(argument_1, argument_2).

// ------------------------------------------------------------
// 1. Types and values
// ------------------------------------------------------------

// A nominal record type gives a fixed shape a reusable name.
type Score {label: String, accepted: Bool}

// Decision is a sum type: every Decision is either Accepted or Rejected.
type Accepted {score: Score}
type Rejected {score: Score, reason: String}
type Decision = Accepted(Accepted) | Rejected(Rejected)

// Constructors belong to their type, so a Decision is always constructed
// through it: Decision.Accepted(Accepted {score: score}). This is the only
// spelling, for local and imported types alike. A constructor with a Unit
// payload takes no arguments, as in Flag.Off(). A generic type's arguments
// are inferred from the payload or the expected type, as in Option.Some(42),
// and can always be written out: Option<Int>.Some(42).

// Entry is reusable for any key and value types.
type Entry<Key, Value> {key: Key, value: Value}
type Compiler {name: String, active: Bool, attempts: Int}
type IndexedNumber {index: Int, value: Int}
type GroupedNumber {group: String, value: Int}
type TourArtifact {
  unit: Unit,
  language: String,
  script: String,
  one_line_script: String,
  compiler: Compiler,
  updated_compiler: Compiler,
  decision: Decision,
  decision_label: String,
  score_entry: Entry<String, Score>,
  answer: Entry<String, Int>,
  answer_label: String,
  outcome: String,
  rendered: List<String>,
  indexed: List<IndexedNumber>,
  filtered: List<Int>,
  repeated: List<Int>,
  copied: List<Int>,
  ordered: List<Int>,
  grouped: Map<String, List<GroupedNumber>>,
  score_type_tag: TypeTag<Score>,
  score_type: TypeDescriptor,
  score_type_id: SemanticTypeId,
  score_schema: TypeSchema,
}

let unit = nil
let enabled = true
let attempts = 3
let ratio = 3.5
let language = "MAG"

// Ordinary quoted strings interpret exactly four escapes: \n, \t, \\, and \".
// Any other backslash escape is a compile error. Triple-quoted strings are
// raw: they may span lines, and quotes, dollar signs, backslashes, and escape-
// like text inside them are preserved literally. They perform no interpolation
// or escape processing; the next three consecutive quotes end the string.
// Use them for embedded shell, regular expressions, Python, JSON, or other
// source text where another language owns the backslashes and quoting.
let script = strip_margin("""|echo 'export PATH="$HOME/.local/bin:$PATH"'
                                |find . \( -name '*.mag' \)""")
let one_line_script = replace(script, "\n", " ")
let yaml_key_pattern = """^  ([^ :]+):\s*$"""
let python_source = strip_margin("""|import re
                                          |line = "  project:"
                                          |print(bool(re.match(r"^  ([^ :]+):\s*$", line)))""")
let numbers = [1, 2, 3]
let compiler = Compiler {name: "mag", active: true, attempts: 3}
let counters: Map<String, Int> = maps.insert(
  maps.insert(maps.empty<String, Int>(), "accepted", 2),
  "rejected",
  1,
)

// Backticks quote punctuation-bearing names when they appear in a qualified
// or other non-infix position. A declared symbolic operator is normally used
// between its operands with whitespace on both sides; call it in prefix form,
// such as (operator)(left, right), only when the surrounding expression needs
// an ordinary function value. A source file and every expression block form a
// block: all let bindings in that block are peers, and the final expression is
// the block's result.

// Record and map access produces ordinary values. Here compiler-name is
// "mag", updated-compiler has attempts 4, and accepted-count is 2.
let compiler_name = get(compiler, "name")
let updated_compiler = assoc(compiler, "attempts", 4)
let accepted_count = maps.get(counters, "accepted")

// ------------------------------------------------------------
// 2. Collections derive new immutable collections
// ------------------------------------------------------------

// first-number is 1 and number-count is 3. Removing position 1 produces
// [1, 3]; the original numbers remain [1, 2, 3]. An invalid position makes
// compilation fail during eager evaluation.
let first_number = first(numbers)
let number_count = count(numbers)
let without_middle = remove_at(numbers, 1)
let joined = concat(numbers, [4, 5])
let field_names = keys(compiler)

// ------------------------------------------------------------
// 3. Typed values, control flow, and functions
// ------------------------------------------------------------

let score = Score {label: "review", accepted: true}
let decision: Decision = if get(score, "accepted") then Decision.Accepted(Accepted {score: score}) else Decision.Rejected(Rejected {score: score, reason: "not accepted"})
let decision_label = match decision {
  case Accepted(accepted) => get(get(accepted, "score"), "label"),
  case Rejected(rejected) => get(rejected, "reason"),
}
let score_entry = Entry<String, Score> {key: "current", value: score}

// A function is an ordinary value. Its body is an expression; a block can
// introduce local bindings and returns its final expression.
let label_score: fn(Score) -> String = |value| => get(value, "label")
let label = label_score(score)

// Generic functions declare type parameters on the binding. Calls may provide
// every argument explicitly or put `_` in a slot the checker should infer.
let entry<Key, Value>: fn(Key, Value) -> Entry<Key, Value> = |key, value| => Entry<Key, Value> {key: key, value: value}
let answer = entry<String, Int>("answer", 42)
let inferred_key = entry<_, Int>("inferred", 7)

let answer_label = str(get(answer, "value"))
let outcome = if enabled then "enabled" else "disabled"
let rendered = map(((|value| => str(value)): fn(Int) -> String), numbers)
let indexed = indexed_map(
  ((|index, value| => IndexedNumber {index: index, value: value}): fn(Int, Int) -> IndexedNumber),
  numbers,
)
let only_two = filter(((|value| => (=)(value, 2)): fn(Int) -> Bool), numbers)
let repeated = flat_map(((|value| => [value, value]): fn(Int) -> List<Int>), numbers)
let copied = fold(
  ((|out, value| => concat(out, [value])): fn(List<Int>, Int) -> List<Int>),
  ([]: List<Int>),
  numbers,
)
let ordered = sort_by(((|value| => str(value)): fn(Int) -> String), [3, 1, 2])
let grouped = group_by(
  ((|item| => get(item, "group")): fn(GroupedNumber) -> String),
  [
    GroupedNumber {group: "b", value: 1},
    GroupedNumber {group: "a", value: 2},
    GroupedNumber {group: "b", value: 3},
  ],
)

// ------------------------------------------------------------
// 4. Types can enter data through checked evidence
// ------------------------------------------------------------

let score_tag = type_tag<Score>()
let score_descriptor = type_evidence(score_tag)
let score_identity = type_id(score_descriptor)
let score_schema = type_schema(score_tag)

// ------------------------------------------------------------
// 5. The artifact
// ------------------------------------------------------------

// artifact is the entry point's result, analogous to the value returned from
// main. It can contain any serializable value. The compiler returns exactly
// this value; MAG adds no application fields to it.
artifact(TourArtifact {
  unit: unit,
  language: language,
  script: script,
  one_line_script: one_line_script,
  compiler: compiler,
  updated_compiler: updated_compiler,
  decision: decision,
  decision_label: decision_label,
  score_entry: score_entry,
  answer: answer,
  answer_label: answer_label,
  outcome: outcome,
  rendered: rendered,
  indexed: indexed,
  filtered: only_two,
  repeated: repeated,
  copied: copied,
  ordered: ordered,
  grouped: grouped,
  score_type_tag: score_tag,
  score_type: score_descriptor,
  score_type_id: score_identity,
  score_schema: score_schema,
})

// Imports are literal and direct. These five forms are distinct:
//
//   import modules.labels                                  // open
//   import modules.labels.{}                               // qualified only
//   import modules.format.{number_label as label}          // selective
//   import modules.data as data                            // aliased namespace
//   import modules.labels.{internal_label as _}            // suppressing
//
// Unqualified visibility is automatic opens - suppressions + positive
// selectors, independent of declaration order. A suppression subtracts only a
// bare import's automatic spelling: qualified access remains, and suppression
// cannot override a positive selector. Dependency visibility is not transitive,
// so every qualified reference requires its own direct import.
//
// Alias-shaped `type Label = String` declarations are transparent: Label and
// String are the same type. `newtype UserId = String` creates a nominal type;
// cross its boundary explicitly with `(raw: UserId)` and `(id: String)`.
// A let annotation alone never coerces a value.
//
// Text and JSON files are immutable source inputs:
//
//   let prompt = read("prompts/reviewer.md")
//   let manifest = read_json("config/toolsets.json")
//   let first_toolset = get(manifest, "toolsets")
```

With Nefor's shipped library, one facade import is enough for a bounded shell
node and its graph artifact:

```mag
import nefor.{ShellScriptParams, Timeout, compile_graph, script}

compile_graph(script("hello", ShellScriptParams {
  script: "printf 'hello\\n'",
  cwd: ".",
  timeout: Timeout.Milliseconds(30000),
}))
```

The compiler writes the artifact itself to stdout. For the four type-evidence
artifacts from the tour, the relevant part of that JSON is:

```json

        { "name": "accepted", "type": { "kind": "primitive", "name": "Bool" } },

```

`score_tag` has static type `TypeTag<Score>` and is the checked witness used by
type-aware built-ins. `score_descriptor` has static type `TypeDescriptor`; it
is the serializable semantic description shown above. `score_identity` is a
`SemanticTypeId`, serialized as a stable SHA-256 string. `score_schema` is a
`TypeSchema`, serialized as the versioned validation schema shown above. A tag
and its evidence currently serialize to the same descriptor shape, but they
are different runtime types and are accepted by different functions.
